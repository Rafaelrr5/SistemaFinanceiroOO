enum PerfilInvestidor {
    CONSERVADOR
    MODERADO
    ARROJADO
}

enum TipoTransacao {
    COMPRA
    VENDA
}

' === INTERFACES ===
interface AtivoNacional <<interface>> {
    +getPrecoEmReais(): double
}

    +converterParaReais(): double
    +getFatorConversao(): double
}

    +cadastrarInvestimento(): boolean
    +podeMovimentar(ativo): boolean
}

' === ATIVO E SUBCLASSES ===
abstract class Ativo {
    -ticker: String
    -nome: String
    -preco: double
    -qualificado: boolean
    +isRendaFixa(): boolean
    +isNacional(): boolean
}

class Acao {
    +getTipoAcao(): String
}
note right of Acao : Ordinária (3)\nPreferencial (4,5,6)\nUnit (11)

class FII {
    -setor: String
    -ultimoDividendo: double
    -taxaAdministracao: double
    +getTaxaAdministracaoFormatada(): String
}

class Tesouro {
    -tipoRendimento: String
    -vencimento: String
}
note right of Tesouro : Renda FIXA

class Stock {
    -bolsaNegociacao: String
    -setor: String
}

class Criptoativo {
    -algoritmoConsenso: String
    -quantidadeMaxima: double
}

' === INVESTIDOR E SUBCLASSES ===
abstract class Investidor {
    -nome: String
    -identificador: String
    -patrimonio: double
    -carteira: Carteira
    +isQualificado(): boolean
}

class PessoaFisica {
    -perfil: PerfilInvestidor
}
note bottom of PessoaFisica : Cripto → só ARROJADO\nStocks → MODERADO/ARROJADO

class Institucional {
    -razaoSocial: String
}
note bottom of Institucional : Pode tudo

' === CARTEIRA ===
class Carteira {
    -ativos: Map<Ativo, ItemCarteira>
    +adicionarAtivo()
    +removerAtivo()
    +getValorTotalCarteira(): double
}

class ItemCarteira {
    -quantidade: double
    -precoMedioCompra: double
    +getValorTotalGasto(): double
    +getValorTotalAtual(): double
}

class Transacao {
    -id: String
    -tipo: TipoTransacao
    -quantidade: double
    -preco: double
    -instituicao: String
}

' === HERANÇA ===
Ativo <|-- Acao

Ativo <|-- Tesouro
Ativo <|-- Stock

Investidor <|-- Institucional

' === INTERFACES ===
Acao ..|> AtivoNacional
FII ..|> AtivoNacional
Tesouro ..|> AtivoNacional
Stock ..|> AtivoInternacional
Criptoativo ..|> AtivoInternacional
Investidor ..|> Investimento

' === ASSOCIAÇÕES ===
Investidor "1" *-- "1" Carteira
Carteira "1" o-- "*" ItemCarteira
ItemCarteira --> Ativo
Transacao --> Ativo
Transacao --> TipoTransacao
PessoaFisica --> PerfilInvestidor
